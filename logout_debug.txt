function logoutUser() {
      DB = getDB();
      DB.currentUser = null;
      saveDB(DB);
      
      const nav = document.getElementById("role-bottom-nav");
      nav.classList.remove("active");
      
      screenHistory = [];
      showScreen("login");
    }
